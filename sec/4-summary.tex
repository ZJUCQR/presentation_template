\section{总结与展望}
% ============================================================
\sectioncover{04}{总结与展望}{}

\begin{frame}{工作总结}
  \slidekey{围绕"数据、奖励、算法、验证"四个环节构建完整后训练闭环。}

  \begin{compactitem}
    \item \textbf{构建物理提示词集：}构建并筛选提示词，覆盖重力碰撞、材料流体等。
    \item \textbf{视觉语言模型双维度奖励：}设计 PC 和 SA 双维度视觉语言模型奖励，分别约束物理常识和语义贴合，减少奖励作弊。
    \item \textbf{轻量级的框架：}实现了轻量级的物理对齐，即不需要物理模拟器，只用LoRA微调0.4\%的参数，完成物理对齐。
    \item \textbf{多维度实验验证：}通过 VBench2.0、人工偏好、消融实验和跨模型实验验证物理后训练有效性。
  \end{compactitem}
\end{frame}

\begin{frame}{未来展望}
  \slidekey{未来可从过程奖励、仿真驱动、表征解耦和自适应训练四个方向推进物理对齐研究。}

  \begin{compactitem}
    \item \textbf{过程奖励建模：}当前奖励机制是结果级的，VLM 对完整视频给出一个标量分数。未来可探索过程奖励建模（PRM），将物理验证分解为逐帧或逐事件评估，以杯子坠落碎裂为例，可分别评估是否离开桌面、下落轨迹是否符合重力、碎裂形态是否合理，提供更密集的信号帮助模型定位和修正具体物理错误。
    \item \textbf{隐空间物理属性解耦：}在视频模型隐空间内将物体的内在物理属性（例如质量、材质）与外在视觉特征（纹理、光影）解耦，通过引入显式物理语义约束，增强零样本物理推理能力和多对象交互场景的生成一致性。
    \item \textbf{多粒度评估与自适应训练：}根据模型在不同物理类别上的表现差异，自适应调整训练策略，对表现较差的类别增加训练强度，进一步提高物理对齐效果。
  \end{compactitem}
\end{frame}
