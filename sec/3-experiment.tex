\begin{frame}{实验设置}
  \slidekey{实验在主模型与跨模型两套设置上验证方法有效性，统一使用 LoRA 低成本适配。}

  \begin{block}{模型与训练配置}
    \begin{compactitem}
      \item 主实验：Wan2.2-TI2V-5B
      \item 跨模型：Wan2.1-T2V-1.3B
      \item 分辨率 $704\times1248$，帧数 81
      \item 采样步数：50
      \item AdamW，学习率 $3\times10^{-5}$
      \item LoRA：rank=32，alpha=64；冻结文本编码器和 VAE
    \end{compactitem}
  \end{block}

  \vspace{0.10cm}
  \begin{exampleblock}{Flow-GRPO 与 VLM 配置}
    \begin{compactitem}
      \item 群组大小 $G=4$
      \item PC 和 SA 权重均为 0.5
      \item 噪声：$\sigma=0.3$
      \item KL 系数 $\beta=10^{-3}$，优势截断 5.0
      \item VLM 评估器：评分范围 1 到 5
    \end{compactitem}
  \end{exampleblock}
\end{frame}
